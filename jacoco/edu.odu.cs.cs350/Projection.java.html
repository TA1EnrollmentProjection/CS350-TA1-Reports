package edu.odu.cs.cs350;

import java.util.ArrayList;
import java.util.Collections;
import java.util.HashMap;
import java.util.List;

/**
 * Projection - Represents the projected number of students that will be enrolled in 
 * a specified course for the current semester based on historic enrollment numbers.
 * 
 * @author David Ortiz
 * @author Mark Foust
 */
public class Projection {
    // Data members
    private final double LAST_DAY = 1.0;
    private HashMap<String, Integer> projections;
    
    public Projection(SemesterEnrollment current, HistoricEnrollment historic, double dayIndex) {
        compute(current, historic, dayIndex);
    }

    public HashMap<String, Integer> getProjections() {
        return projections;
    }
    
    private void compute(SemesterEnrollment current, HistoricEnrollment historic, double dayIndex) {
        List<String> courseNames = getAllCourseNames(current.getSnapshots());
        Integer[] lastDayEnrollments = getHistoricEnrollments(courseNames, historic, LAST_DAY);
        Integer[] specificDayEnrollments = getHistoricEnrollments(courseNames, historic, dayIndex);
        Integer[] currentEnrollments = getCourseEnrollments(current, courseNames, dayIndex);
        Integer[] projectionNumbers = computeProjections(lastDayEnrollments, specificDayEnrollments, currentEnrollments);
        projections = mergeNameAndProjection(courseNames, projectionNumbers);
    }

    private List<String> getAllCourseNames(List<SemesterSnapshot> snapshots) {
        List<String> courseNames = new ArrayList<>();
        for (SemesterSnapshot snap : snapshots) {
            List<Course> courseList = snap.getCourseList();
            courseList.forEach((course) -> {
                String courseName = course.getCourseName();
                if (!courseNames.contains(courseName)) {
                    courseNames.add(courseName);
                }
            });
        }
        Collections.sort(courseNames);
        return courseNames;
    }

    private Integer[] getHistoricEnrollments(List<String> current, HistoricEnrollment historic, double dayIndex) {
        int numberOfSemesters = historic.getNumberOfSemesters();
        int numberOfCourses = current.size();
        List<SemesterEnrollment> semesters = historic.getSemesters();
        Integer[][] lastDayEnrollments = new Integer[numberOfSemesters][numberOfCourses];
        for (int semester = 0; semester < numberOfSemesters; semester++) {
            lastDayEnrollments[semester] = getCourseEnrollments(semesters.get(semester), current, dayIndex);
        }
        return averageEnrollments(lastDayEnrollments);
    }

    private Integer[] getCourseEnrollments(SemesterEnrollment semester, List<String> current, double dayIndex) {
        Integer[] enrollments = new Integer[current.size()];
        List<SemesterSnapshot> snapshots = semester.getSnapshots();
        int desiredDate = (int) ((snapshots.size() - 1) * dayIndex);
        SemesterSnapshot desiredSnapshot = snapshots.get(desiredDate);
        for (int course = 0; course < current.size(); course++) {
            int courseIndex = desiredSnapshot.findCourse(current.get(course));
            if (courseIndex >= 0) {
                Course desiredCourse = desiredSnapshot.getCourse(courseIndex);
                enrollments[course] = desiredCourse.getNumEnrolledStudents();
            } else {
                enrollments[course] = 0;
            }
        }
        return enrollments;
    }

    private Integer[] averageEnrollments(Integer[][] enrollments) {
        int numberOfEnrollments = enrollments.length;
        Integer[] averagedEnrollments = new Integer[enrollments[0].length];
        for (int index = 0; index < averagedEnrollments.length; index++) {
            int sum = sumEnrollments(enrollments, index);
            int averaged = sum / numberOfEnrollments;
            averagedEnrollments[index] = averaged;
        }
        return averagedEnrollments;
    }

    private int sumEnrollments(Integer[][] enrollments, int index) {
        int sumTotal = 0;
        for (int i = 0; i < enrollments.length; i++) {
            sumTotal += enrollments[i][index];
        }
        return sumTotal;
    }

    private Integer[] computeProjections(Integer[] lastDay, Integer[] targetDay, Integer[] current) {
        Integer[] projections = new Integer[current.length];
        for (int index = 0; index < projections.length; index++) {
            double projection = (double) (current[index] * lastDay[index]) / targetDay[index];
            projections[index] = (int) Math.round(projection);
        }
        return projections;
    }

    private HashMap<String, Integer> mergeNameAndProjection(List<String> names, Integer[] numbers) {
        HashMap<String, Integer> tempNumbers = new HashMap<>();
        for (int i = 0; i < names.size(); i++) {
            tempNumbers.put(names.get(i), numbers[i]);
        }
        return tempNumbers;
    }
