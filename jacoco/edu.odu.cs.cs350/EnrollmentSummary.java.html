package edu.odu.cs.cs350;

public class EnrollmentSummary {

    private String courseName;
    private int currentEnrollment;
    private int projectedEnrollment;
    private int totalCap;

    public EnrollmentSummary(String courseName,
                             int currentEnrollment,
                             int projectedEnrollment,
                             int totalCap) {
        this.courseName = courseName;
        this.currentEnrollment = currentEnrollment;
        this.projectedEnrollment = projectedEnrollment;
        this.totalCap = totalCap;
    }

    public String getCourseName() {
        return courseName;
    }

    public int getCurrentEnrollment() {
        return currentEnrollment;
    }

    public int getProjectedEnrollment() {
        return projectedEnrollment;
    }

    public int getTotalCap() {
        return totalCap;
    }

    public boolean isOverCapacity() {
        return projectedEnrollment > totalCap;
    }

    public char getMarker() {
        return isOverCapacity() ? '*' : ' ';
    }

    public static String summaryReport(SemesterSnapshot semester, int percent) {
        final String header = " Course  Enrollment  Projected  Cap";
        StringBuilder outSB = new StringBuilder();
        outSB.append(String.format("%s%% of enrollment period has elapsed.%n", percent));
        outSB.append(String.format("%s%n", header));
        for (Course course : semester.getCourseList()) {
            outSB.append(String.format(" %6s  %10d  %9s  %d%n", course.getCourseName(), course.getNumEnrolledStudents(), "-", course.getTotalEnrolledCapacity()));
        }
        return outSB.toString();
    }

}
