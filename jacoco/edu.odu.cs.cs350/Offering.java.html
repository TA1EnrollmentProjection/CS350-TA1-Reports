package edu.odu.cs.cs350;

import java.util.ArrayList;
import java.util.List;

/**
* Represents an offering with a list of sections, the total number of students enrolled, and the offering's capacity.
**/
public class Offering implements Cloneable {
    private int numEnrolled;
    private int offeringCapacity;
    private List<Section> sections;  

    /**
     * Default constructor for Offering
     */
    public Offering(){
        this.numEnrolled = 0;  
        this.offeringCapacity = 0; 
        this.sections = new ArrayList<>(); 
    }

    /**
     * Full constructor for Offering with manual input for capacity and enrollment
     * @param sections
     * @param offeringCapacity
     * @param numEnrolled
     */
    public Offering(List<Section> sections, int offeringCapacity, int numEnrolled) {
        this.numEnrolled = numEnrolled;
        this.offeringCapacity = offeringCapacity;
        this.sections = sections; 

    }

    /**
     * constructor for Offering with manual input for capacity, automatic calculation of enrollment
     * @param sections
     * @param offeringCapacity
     */
    public Offering(List<Section> sections, int offeringCapacity) {
        numEnrolled = 0; 
        this.offeringCapacity = offeringCapacity;
        this.sections = sections;
        totalSectionEnrollment();
    }

    /**
     * constructor for Offering with automatic calculation of capacity and enrollment
     * @param sections
     */
    public Offering(List<Section> sections) {
        this.sections = sections;
        totalSectionCapacity();
        totalSectionEnrollment();
    }

    public int getNumEnrolled(){
        return numEnrolled;
    }

    public void setNumEnrolled(int numEnrolled){
        this.numEnrolled = numEnrolled;
    }

    public int getOfferingCapacity(){
        return offeringCapacity;
    }

    public void setOfferingCapacity(int offeringCapacity){
        this.offeringCapacity = offeringCapacity;
    }

    public List<Section> getSections(){
        return sections; 
    }

    /**
     * Calculates the enrollment of an Offering by summing the enrollment of all sections.
     */
    private void totalSectionEnrollment(){
        int sum = 0;
        for(Section section : sections) {
            sum += section.getNumEnrolled();
        }
        if (sum > offeringCapacity) {
            System.out.println("Number enrolled is greater than capacity");
            setNumEnrolled(offeringCapacity);
        } else {
            setNumEnrolled(sum);
        }
    }

    /**
     * Calculates the capacity of an Offering by summing the capacity of all sections.
     */
    private void totalSectionCapacity(){
        int sum = 0;
        for(Section section : sections) {
            sum += section.getCrossListCapacity();
        }
        setOfferingCapacity(sum);
    

    }

    @Override
    public Offering clone() {
        List<Section> cloneSection = new ArrayList<>();
        for (Section sect : sections) {
            cloneSection.add(sect.clone());
        }
        return new Offering(cloneSection, this.offeringCapacity, this.numEnrolled);
    }

    @Override 
    public boolean equals(Object obj){
        if (obj instanceof Offering) {
            Offering o = (Offering) obj;
            if (!sections.equals(o.sections))
                return false;
            return offeringCapacity == o.offeringCapacity && numEnrolled == o.numEnrolled;
        } else
            return false;
    }

    @Override 
    public int hashCode() {
        int result = 0;
        for(Section section : sections) {
            result += section.hashCode();
        }
        result += Integer.hashCode(offeringCapacity);
        result += Integer.hashCode(numEnrolled);
        return result;
    }

    @Override
    public String toString(){
        String offeringString = ("number enrolled: " + numEnrolled);
        offeringString += ("\n" + "offering capacity: " + offeringCapacity);
        offeringString += ("\n" +"offering name: " + sections.get(0).getCrossListGroup()); 
        return offeringString; 
    }
    

}
