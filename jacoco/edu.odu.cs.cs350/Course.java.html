package edu.odu.cs.cs350;

import java.util.ArrayList;
import java.util.List;

/**
 * Represents a course offered by ODU with the course name, number of enrolled students,
 * total capacity of enrolled students, and the list of course offerings.
 * 
 * @author David Ortiz
 */
public class Course implements Comparable<Course>, Cloneable {
    // Data members
    private final String courseName;
    private final int numEnrolledStudents;
    private final int totalEnrolledCapacity;
    private final List<Offering> courseOfferings;

    /**
     * Default constructor for Course.
     */
    public Course() {
        this.courseName = "";
        this.numEnrolledStudents = 0;
        this.totalEnrolledCapacity = 0;
        this.courseOfferings = new ArrayList<>();
    }

    /**
     * Primary full constructor for Course.
     * 
     * @param courseName the name of the course
     * @param courseOfferings a List of Offerings comprising the Course
     */
    public Course(String courseName, List<Offering> courseOfferings) {
        this.courseName = courseName;
        this.courseOfferings = cloneOfferings(courseOfferings);
        this.totalEnrolledCapacity = sumTotalCapacityInOfferings(this.courseOfferings);
        this.numEnrolledStudents = sumEnrolledInOfferings(this.courseOfferings);
    }

    /** 
     * Secondary full constructor for Course.
     * 
     * @param courseName the name of the course
     * @param numEnrolledStudents the number of students enrolled
     * @param totalEnrolledCapacity the max capacity of enrolled students
     * @param courseOfferings a List of Offerings comprising the Course
     */
    public Course(String courseName, 
                  int numEnrolledStudents, 
                  int totalEnrolledCapacity, 
                  List<Offering> courseOfferings) {
        this.courseName = courseName;
        this.numEnrolledStudents = numEnrolledStudents;
        this.totalEnrolledCapacity = totalEnrolledCapacity;
        this.courseOfferings = courseOfferings;
    }

    private List<Offering> cloneOfferings(List<Offering> courseOfferings) {
        List<Offering> cloned = new ArrayList<>();
        for (Offering offer : courseOfferings) {
            cloned.add(offer.clone());
        }
        return cloned;
    }

    private int sumEnrolledInOfferings(List<Offering> offerings) {
        int sum = 0;
        for (Offering offering : offerings) {
            sum += offering.getNumEnrolled();
        }
        // Safeguard against more enrolled than capacity
        // May remove this check later
        if (sum > totalEnrolledCapacity) {
            return totalEnrolledCapacity;
        } else {
            return sum;
        }   
    }

    private int sumTotalCapacityInOfferings(List<Offering> offerings) {
        int sum = 0;
        for (Offering offering : offerings) {
            sum += offering.getOfferingCapacity();
        }
        return sum;
    }

    /**
     * Get the number of enrolled students in the Course.
     * 
     * @return the number of enrolled students
     */
    public int getNumEnrolledStudents() {
        return numEnrolledStudents;
    }

    /**
     * Get the name of the course.
     * 
     * @return the name of the course
     */
    public String getCourseName() {
        return courseName;
    }

    /** 
     * Get the total enrolled capacity of the course.
     * 
     * @return the total enrolled capacity
     */
    public int getTotalEnrolledCapacity() {
        return totalEnrolledCapacity;
    }

    /**
     * Compares two Course objects for sorting.
     * 
     * @see java.lang.Comparable#compareTo(java.lang.Object)
     */
    @Override
    public int compareTo(Course other) {
        return this.courseName.compareTo(other.courseName);
        
    }

    /**
     * Provides a deep copy of the Course object.
     * 
     * @see java.lang.Object#clone()
     */
    @Override
    public Course clone() {
        List<Offering> cloneOfferings = new ArrayList<>();
        for (Offering offer : courseOfferings) {
            cloneOfferings.add(offer.clone());
        }
        return new Course(this.courseName, this.numEnrolledStudents, 
                          this.totalEnrolledCapacity, cloneOfferings);
    }

    /**
     * Compares two Course objects for equality based on 
     * course name and the number of enrolled students
     * 
     * @see java.lang.Object#equals(java.lang.Object)
     */
    @Override
    public boolean equals(Object obj) {
        if (this == obj) {
            return true;
        }
        if (obj == null || getClass() != obj.getClass()) {
            return false;
        }
        Course course = (Course) obj;
        return courseName.equals(course.courseName) && 
               numEnrolledStudents == course.numEnrolledStudents;
    }

    /**
     * Computes the hashcode value of a Course object
     * 
     * @see java.lang.Object#hashCode()
     */
    @Override
    public int hashCode() {
        int result = courseName.hashCode();
        result = (31 * result) + (373 * numEnrolledStudents);
        return result;
    }

    /**
     * Gives a String displaying course name, number of 
     * enrolled students, and total capacity of students
     * 
     * @see java.lang.Object#toString()
     */
    @Override
    public String toString() {
        return String.format("Course Name: %s\tEnrolled Students: %d\tCapacity: %d%n", 
                             courseName, numEnrolledStudents, totalEnrolledCapacity);
    }
}
