package edu.odu.cs.cs350;

import java.util.ArrayList;
import java.util.List;

/**
 * SemesterSnapshot - Represents snapshot of enrollment data (courses, 
 * offerings, and sections) for a list of (CS) courses on a specified date.
 * 
 * @author David Ortiz
 */
public class SemesterSnapshot {
    // Data members
    private final String date;
    private final List<Course> courseList;

    /**
     * Default constructor for SemesterSnapshot.
     */
    public SemesterSnapshot() {
        this.date = "";
        this.courseList = new ArrayList<>();
    }

    /**
     * Full constructor for SemesterSnapshot.
     * 
     * @param date the date associated with the snapshot
     * @param courseList a List of Courses comprising the snapshot
     */
    public SemesterSnapshot(String date, List<Course> courseList) {
        this.date = date;
        this.courseList = new ArrayList<>();
        for (int i = 0; i < courseList.size(); i++) {
            this.courseList.add(courseList.get(i).clone());
        }
    }

    /**
     * Get the date associated with the snapshot.
     * 
     * @return the date in YYYY-MM-DD format
     */
    public String getDate() {
        return date;
    }

    /**
     * Get the List of courses for this snapshot.
     * 
     * @return cloned copy of the course list
     */
    public List<Course> getCourseList() {
        List<Course> copyList = new ArrayList<>();
        for (Course course : courseList) {
            copyList.add(course.clone());
        }
        return copyList;
    }

    public List<String> getCourseNameList() {
        List<String> courseNames = new ArrayList<>();
        for (Course course : courseList) {
            courseNames.add(course.getCourseName());
        }
        return courseNames;
    }

    public int getSnapshotSize() {
        return courseList.size();
    }

    /**
     * Gets a course from the course list at a specified index
     * 
     * @param index the index of the specified course
     * @return the specified course
     */
    public Course getCourse(int index) {
        return courseList.get(index).clone();
    }

    /**
     * Search the course list by course name. Returns a (-1) if not found.
     * 
     * @param courseName the name of the course
     * @return the index in the list of the course
     */
    public int findCourse(String courseName) {
        int index = -1;
        for (Course course : courseList) {
            if (course.getCourseName().equals(courseName)) {
                index = courseList.indexOf(course);
            }
        }
        return index;
    }

    /**
     * Compares two SemesterSnapshot objects for equality 
     * based on the date and the list of courses.
     * 
     * @see java.lang.Object#equals(java.lang.Object)
     */
    @Override 
    public boolean equals(Object obj) {
        if (this == obj) {
            return true;
        }
        if (obj == null || getClass() != obj.getClass()) {
            return false;
        }
        SemesterSnapshot semSnap = (SemesterSnapshot) obj;
        return date.equals(semSnap.date) && (courseList.equals(semSnap.courseList));
    }

    /**
     * Computes the hashcode value for the SemesterSnaphot object.
     * 
     * @see java.lang.Object#hashCode()
     */
    @Override
    public int hashCode() {
        int hash = 0;
        hash += (37 * date.hashCode());
        for (Course course : courseList) {
            hash += (373 * course.hashCode());
        }
        return hash;
    }

    /**
     * Gives a String displaying the date associated with 
     * the SemesterSnapshot object and a list of its courses. 
     * 
     * @see java.lang.Object#toString()
     */
    @Override
    public String toString() {
        StringBuilder semSnapSB = new StringBuilder();
        semSnapSB.append(String.format("Semester Snapshot%n"));
        semSnapSB.append(String.format("Date: %s%nCourses:%n", this.date));
        for (Course course : courseList) {
            semSnapSB.append(String.format("%s%n", course.getCourseName()));
        }
        return semSnapSB.toString();
    }
}
