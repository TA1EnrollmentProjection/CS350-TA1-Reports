package edu.odu.cs.cs350; 

/**
* Represents a course section with the offering it's under (crossListGroup), the number enrolled,
* the capacity of the section, and the CRN.
**/
public class Section implements Cloneable {
    private String crossListGroup;
    private int numEnrolled;
    private int crossListCapacity;
    private String CRN;

    /**
    * Default constructor for Section
    **/
    public Section(){
        this.crossListGroup = "";
        this.numEnrolled = 0;
        this.crossListCapacity = 0; 
        this.CRN = "";
    }

    /**
    * Full constructor for section
    **/
    public Section(String crossListGroup, int numberEnrolled, int crossListCapacity, String CRN) {
        this.crossListGroup = crossListGroup;
        this.numEnrolled = numberEnrolled;
        this.crossListCapacity = crossListCapacity;
        this.CRN = CRN;
    }

    public String getCrossListGroup(){
        return crossListGroup; 
    }

    public void setCrossListGroup(String crossListGroup){
        this.crossListGroup = crossListGroup; 
    }

    public int getNumEnrolled(){
        return numEnrolled;
    }

    public void setNumEnrolled(int numberEnrolled){
        this.numEnrolled = numberEnrolled;
    }

    public int getCrossListCapacity(){
        return crossListCapacity;
    }

    public void setCrossListCapacity(int crossListCapacity){
        this.crossListCapacity = crossListCapacity; 
    }

    public String getCRN(){
        return CRN; 
    }

    public void setCRN(String CRN){
        this.CRN = CRN; 
    }

    @Override 
    public boolean equals(Object obj) {
        if (obj instanceof Section) {
            Section s = (Section) obj;
            if (!CRN.equals(s.CRN))
                return false;
            return CRN == s.CRN;
        } else
            return false;
    }

    @Override 
    public int hashCode() {
        return CRN.hashCode();
    }

    @Override
    public String toString(){
        String sectionString = ("crossListGroup: " + crossListGroup);
        sectionString += ("\n" + "number enrolled: " + numEnrolled);
        sectionString += ("\n" + "total capacity: " + crossListCapacity);
        sectionString += ("\n" + "CRN: " + CRN);
        return sectionString;
    }

    @Override
    public Section clone() {
        Section clone = new Section(this.crossListGroup, this.numEnrolled, this.crossListCapacity, this.CRN);
        return clone;
    }
   
}
