package edu.odu.cs.cs350;

import java.io.File;
import java.io.IOException;
import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class SemesterEnrollment {
    private String semesterCode;
    private List<SemesterSnapshot> snapshots;

    public SemesterEnrollment(){
        this.semesterCode = "";
        this.snapshots = new ArrayList<>();
    }

    public SemesterEnrollment(String directoryPath) throws IOException {
        int directoryNameIndex = directoryPath.lastIndexOf("/") + 1;
        this.semesterCode = directoryPath.substring(directoryNameIndex);
        this.snapshots = new ArrayList<>();
        buildSemesterEnrollment(directoryPath);
    }

    public String getSemesterCode() {
        return semesterCode;
    }

    public List<SemesterSnapshot> getSnapshots() {
        return snapshots;
    }

    public void setSemesterCode(String semesterCode) {
        this.semesterCode = semesterCode;
    }

    private void buildSemesterEnrollment(String directoryPath) throws IOException {

        File directory = new File(directoryPath);
        File[] directoryFiles = directory.listFiles();
        
        String startDate = "";
        String endDate = "";

        for(File file : directoryFiles){
            if(file.isFile() && file.getName().equals("dates.txt")){
                //I'll handle this handling later
                Scanner fileScanner = new Scanner(file);
                startDate = fileScanner.nextLine();
                endDate = fileScanner.nextLine();
                fileScanner.close();
            }
        }

        selectFilesInDate(startDate, endDate, directoryFiles);
        
    }

    private void selectFilesInDate(String startDate, String endDate, File[] directoryFiles) throws IOException {
        for(File file : directoryFiles){
            String fileName = file.getName().replace(".csv", "");
            if(file.isFile() && !file.getName().contains("txt") && betweenDates(fileName, startDate, endDate)){
                //More error handling for another day
                SemesterSnapshot snapshot = SnapshotReader.readSemesterSnapshot(file, fileName);
                snapshots.add(snapshot);
            }
        }
    }

    private boolean betweenDates(String fileName, String startDate, String endDate){
        int fileDate = Integer.parseInt(fileName.replace("-", ""));
        int startNum = Integer.parseInt(startDate.replace("-", ""));
        int endNum = Integer.parseInt(endDate.replace("-", ""));
        return fileDate >= startNum && fileDate <= endNum;
    }

    
}
