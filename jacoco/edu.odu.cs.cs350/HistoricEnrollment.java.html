package edu.odu.cs.cs350;

import java.io.File;
import java.util.List;
import java.util.ArrayList;
import java.util.Collections;
import java.util.HashMap;


public class HistoricEnrollment {
    // Data Members
    private List<SemesterEnrollment> semesters;

    /**
     * Default constructor for HistoricEnrollment.
     */
    public HistoricEnrollment() {
        semesters = new ArrayList<SemesterEnrollment>();
    }

    /**
     * Full constructor for HistoricEnrollment.
     */
    public HistoricEnrollment(String[] directories) {
        semesters = new ArrayList<SemesterEnrollment>();
        for (String directory : directories) {
            SemesterEnrollment semester = buildHistoricEnrollment(directory);
            semesters.add(semester);
        }
    }

    /**
     * Builds a SemesterEnrollment from a directory.
     */
    private SemesterEnrollment buildHistoricEnrollment(String directory) {
        SemesterEnrollment semester = null;
        try {
            semester = new SemesterEnrollment(directory);
        } catch (Exception e) {
            System.err.println(e);
        }
        return semester;
    }

    /**
     * Get the number of semesters of historic data.
     * 
     * @return number of semesters
     */
    public int getNumberOfSemesters() {
        return semesters.size();
    }

    /**
     * Gets the list of SemesterEnrollments held in the HistoricEnrollment.
     * 
     * @return List of SemesterEnrollments
     */
    public List<SemesterEnrollment> getSemesters() {
        return semesters;
    }

    /**
     * Checks if data for a specified semester exists in the HistoricEnrollment.
     * 
     * @param semesterName the semester code for the data
     * @return true if found, otherwise false
     */
    public boolean contains(String semesterName) {
        for (SemesterEnrollment semester : semesters) {
            if (semester.getSemesterCode().equals(semesterName)) {
                return true;
            }
        }
        return false;
    }

    /**
     * Computes projections for enrollment based on the historic enrollment data.
     * 
     * @return a map of courses to projected enrollment number
     */
    public HashMap<String, Integer> computeProjections() {
        String[] courseNames = getAllCourseNames();
        Integer[] projectionNumbers = projectionComputations(courseNames.length);
        HashMap<String, Integer> projections = mergeNameAndProjection(courseNames, projectionNumbers);
        return projections;
    }

    private String[] getAllCourseNames() {
        String[] courseNames = new String[0];
        for (SemesterEnrollment semester : semesters) {
            String[] tempNames = filterAllCourseNames(semester.getSnapshots());
            if (tempNames.length > courseNames.length) {
                courseNames = tempNames;
            }
        }
        return courseNames;
    }

    private String[] filterAllCourseNames(List<SemesterSnapshot> snapshots) {
        List<String> courseNames = new ArrayList<>();
        for (SemesterSnapshot snapshot: snapshots) {
            List<String> tempList = snapshot.getCourseNameList();
            if (tempList.size() > courseNames.size()) {
                courseNames = tempList;
            }
        }
        return courseNames.toArray(new String[courseNames.size()]);
    }

    private Integer[] projectionComputations(int numberOfCourses) {
        Integer[] projectionNumbers = new Integer[numberOfCourses];
        for (int i = 0; i < projectionNumbers.length; i++) {
            projectionNumbers[i] = 0;
        }
        return projectionNumbers;
    }

    private HashMap<String, Integer> mergeNameAndProjection(String[] names, Integer[] numbers) {
        HashMap<String, Integer> tempNumbers = new HashMap<>();
        for (int i = 0; i < names.length; i++) {
            tempNumbers.put(names[i], numbers[i]);
        }
        return tempNumbers;
    }
}
