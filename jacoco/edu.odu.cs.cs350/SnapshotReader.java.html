package edu.odu.cs.cs350;

import java.io.File; 
import java.io.IOException;
import java.nio.file.Files;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Map;
import java.util.LinkedHashMap;
import java.util.List;

/**
 * Locates and reads enrollment snapshot files.
 * 
 * Finds CSV snapshot files from a given directory,
 * reads their rows, and converts them into Section objects.
 */
public class SnapshotReader {

    // CSV Column Positions
    private static final int CRN = 1;
    private static final int COURSE_NAME = 2;
    private static final int COURSE_NUMBER = 3;
    private static final int COURSE_TITLE = 4;
    private static final int CROSS_LIST_CAPACITY = 6;
    private static final int NUM_ENROLLED = 7;
    private static final int CROSS_LIST_GROUP = 9;
    private static final int OVERALL_CAPACITY = 22;

    /**
     * Validates the command-line arguments and lists
     * CSV files in a specified directory.
     *
     * @param args command-line arguments containing
     *             path to the directory
     * @return 0 if arguments are valid, 1 otherwise
     */
    public static int run(String[] args) {
        String directoryPath = getDirectory(args);

        if (directoryPath == null || directoryPath.isBlank()) {
            System.err.println("Error: No directory provided.");
            return 1;
        }

        File directory = new File(directoryPath);

        if (!directory.isDirectory()) {
            System.err.println("Error: Invalid directory.");
            return 1;
        }

        File[] csvFiles = getCSVFiles(directory);

        if (csvFiles == null) {
            System.err.println("Error: Unable to retrieve CSV files.");
            return 1;
        }

        for (File file : csvFiles) {
            System.out.println(file.getName());
        }

        return 0;
    }

    /**
     * Obtains a directory path provided through
     * command-line arguments.
     *
     * @param args command-line arguments containing
     *             a directory path
     * @return a directory path, or null if no
     *         directory was provided
     */
    public static String getDirectory(String[] args) {
        if (args.length == 0) {
            return null;
        }

        return args[0];
    }

    /**
     * Retrieves files and subdirectories contained
     * within a specified directory.
     *
     * @param directory to retrieve contents from
     * @return an array containing the contents of a directory,
     *         or null if the directory is invalid or unreadable
     */
    public static File[] getDirectoryContents(File directory) {
        if (directory == null || !directory.isDirectory()) {
            return null;
        }

        return directory.listFiles();
    }

    /**
     * Retrieves CSV files from a specified directory.
     *
     * @param directory containing CSV files
     * @return an array containing CSV files, or null
     *         if the directory is invalid or unreadable
     */
    public static File[] getCSVFiles(File directory) {
        File[] contents = getDirectoryContents(directory);

        if (contents == null) {
            return null;
        }

        File[] csvFiles = new File[contents.length];
        int count = 0;

        for (File file : contents) {
            if (file.isFile() && file.getName().endsWith(".csv")) {
                csvFiles[count] = file;
                count++;
            }
        }

        File[] result = Arrays.copyOf(csvFiles, count);
        Arrays.sort(result);

        return result;
    }

    /**
     * Retrieves the grouping key for a section and
     * uses the CRN if section has no cross-list group.
     *
     * @param section to retrieve the group key from
     * @return the cross-list group or CRN
     */
    public static String getGroupKey(Section section) {
        String group = section.getCrossListGroup();

        if (group == null || group.isBlank()) {
            return section.getCRN();
        }

        return group;
    }

    /**
     * Retrieves a CSV snapshot file for a specified date.
     *
     * @param directory containing CSV files
     * @param date of the requested snapshot
     * @return the matching CSV file, or null if
     *         no matching file is found
     */
    public static File getSnapshotFile(File directory, String date) {
        File[] csvFiles = getCSVFiles(directory);

        if (csvFiles == null || date == null) {
            return null;
        }

        String fileName = date + ".csv";

        for (File file : csvFiles) {
            if (file.getName().equals(fileName)) {
                return file;
            }
        }

        return null;
    }

    /**
     * Parses a CSV line into a Section object.
     *
     * @param fields the fields of the CSV file
     * @return a Section object containing enrollment data
     */
    public static Section parseSection(String[] fields) {
        if (fields == null || fields.length <= CROSS_LIST_GROUP) {
            throw new IllegalArgumentException("Expected at least " + (CROSS_LIST_GROUP + 1) + " fields.");
        }

        String crossListGroup = fields[CROSS_LIST_GROUP];
        int numEnrolled = Integer.parseInt(fields[NUM_ENROLLED].trim());
        int crossListCapacity = Integer.parseInt(fields[CROSS_LIST_CAPACITY].trim());
        String crn = fields[CRN];

        return new Section(crossListGroup, numEnrolled, crossListCapacity, crn);
    }

    /**
     * Groups sections by their cross-list group.
     * Sections without a cross-list group are grouped by CRN.
     *
     * @param sections the sections to organize
     * @return a map containing grouped sections
     */
    public static Map<String, List<Section>> groupSectionsByCrossList(
            List<Section> sections) {

        Map<String, List<Section>> groups = new LinkedHashMap<>();

        for (Section section : sections) {
            groups.computeIfAbsent(getGroupKey(section), key -> new ArrayList<>()).add(section);
        }

        return groups;
    }

    /**
     * Reads section data from a CSV snapshot and 
     * converts rows into Section objects.
     *
     * @param file the CSV snapshot file
     * @return a list of sections from the snapshot
     * @throws IOException if the file cannot be read
     */
    public static List<Section> readSnapshot(File file) throws IOException {
        List<Section> sections = new ArrayList<>();
        List<String[]> rows = readSnapshotRows(file);

        for (String[] fields : rows) {
            sections.add(parseSection(fields));
        }

        return sections;
    }

    /**
     * Reads the rows of a CSV snapshot. Header row 
     * and blank lines are ignored.
     *
     * @param file the CSV snapshot file
     * @return a list of rows from the snapshot
     * @throws IOException if an error occurs when reading the file
     */
    public static List<String[]> readSnapshotRows(File file) throws IOException {
        List<String[]> rows = new ArrayList<>();
        List<String> lines = Files.readAllLines(file.toPath());

        for (int i = 1; i < lines.size(); i++) {
            String line = lines.get(i);

            if (line.isBlank()) {
                continue;
            }

            // Split on commas that are outside of double quotes
            String[] fields = line.split(",(?=(?:[^\"]*\"[^\"]*\")*[^\"]*$)", -1);

            for (int j = 0; j < fields.length; j++) {
                fields[j] = fields[j].replaceAll("^\"|\"$", "");
            }
            if (isExcludedSection(fields)){
                continue;
            }
            rows.add(fields);
        }

        return rows;
    }

    /**
     * Extracts the course identifier from CSV fields.
     * 
     * @param fields the fields of the CSV file 
     * @return the course identifier
     * @throws IllegalArgumentException if the fields array is null or has elements less than or equal to COURSE_NUMBER
     */
    public static String getCourseIdentifier(String[] fields) {
        if (fields == null || fields.length <= COURSE_NUMBER) {
            throw new IllegalArgumentException("Expected at least " + (COURSE_NUMBER + 1) + " fields.");
        }

        return fields[COURSE_NAME] + fields[COURSE_NUMBER];
    }

    /**
     * Extracts the course title from CSV fields.
     * 
     * @param fields the fields of the CSV file
     * @return the course title
     * @throws IllegalArgumentException if the fields array is null or has elements less than or equal to COURSE_TITLE
     */
    public static String getCourseTitle(String[] fields) {
        if (fields == null || fields.length <= COURSE_TITLE) {
            throw new IllegalArgumentException("Expected at least " + (COURSE_TITLE + 1) + " fields.");
        }

        return fields[COURSE_TITLE];
    }

    /**
     * Finds existing labs and recitations within CSV title field
     * and extracts the course title without lab or recitation information.
     */
    public static boolean isExcludedSection(String[] fields) {
        String title = getCourseTitle(fields).toUpperCase();
        if (title.contains(" LAB") || title.contains(" REC")) {
            return true;
        }
        return false;
    }

    /**
     * Retrieves the overall offering capacity from a CSV record.
     *
     * @param fields the CSV record fields
     * @return the overall offering capacity
     * @throws IllegalArgumentException if the record is invalid
     */
    public static int getOverallCapacity(String[] fields) {
        if (fields == null || fields.length <= OVERALL_CAPACITY) {
            throw new IllegalArgumentException("Expected at least " + (OVERALL_CAPACITY + 1) + " fields.");
        }

        return Integer.parseInt(fields[OVERALL_CAPACITY].trim());
    }

    /**
     * Creates an Offering from a list of sections.
     *
     * @param sections the sections belonging to the offering
     * @param overallCapacity the overall offering capacity
     * @return the constructed Offering object
     */
    public static Offering createOffering(List<Section> sections, int overallCapacity) {

        return new Offering(sections, overallCapacity);
    }

    /**
     * Determines whether a CSV record has a value in the
     * overall capacity column.
     *
     * @param fields the CSV record fields
     * @return true if the overall capacity is present and not blank
     */
    private static boolean hasOverallCapacity(String[] fields) {

        return fields.length > OVERALL_CAPACITY && !fields[OVERALL_CAPACITY].isBlank();
    }

    /**
     * Creates Offering objects from CSV snapshot records.
     *
     * @param rows the CSV records containing section information
     * @return a list of Offering objects
     */
    public static List<Offering> createOfferingsFromRows(List<String[]> rows) {
        Map<String, List<Section>> groups = new LinkedHashMap<>();
        Map<String, Integer> capacities = new LinkedHashMap<>();
 
        for (String[] fields : rows) {
            Section section = parseSection(fields);
            String group = getGroupKey(section);
 
            List<Section> members = groups.get(group);
 
           if (members == null) {
                members = new ArrayList<>();
                groups.put(group, members);
                capacities.put(group, section.getCrossListCapacity());
            }

            if (hasOverallCapacity(fields)) {
                capacities.put(group, getOverallCapacity(fields));
            }

            members.add(section);
        }

        List<Offering> offerings = new ArrayList<>();

        for (String group : groups.keySet()) {
            offerings.add(createOffering(groups.get(group), capacities.get(group)));
        }

        return offerings;
    }

    /**
     * Creates a Course from its name and associated Offerings.
     *
     * @param courseName the name of the course
     * @param offerings the Offerings belonging to the course
     * @return the constructed Course object
     */
    public static Course createCourse(String courseName, List<Offering> offerings) {

        return new Course(courseName, offerings);
    }

    /**
     * Creates Course objects from CSV snapshot data.
     *
     * @param rows the CSV records containing course information
     * @return a list of Course objects
     */
    public static List<Course> createCoursesFromRows(List<String[]> rows) {
        Map<String, List<String[]>> courseRows = new LinkedHashMap<>();

        for (String[] fields : rows) {
            String courseName = getCourseIdentifier(fields);
            courseRows.computeIfAbsent(courseName, key -> new ArrayList<>()).add(fields);
        }

        List<Course> courses = new ArrayList<>();

        for (String courseName : courseRows.keySet()) {
            List<Offering> offerings =createOfferingsFromRows(courseRows.get(courseName));
            courses.add(createCourse(courseName, offerings));
        }

        return courses;
    }

    /**
     * Creates a SemesterSnapshot from a date and list of courses.
     *
     * @param date of the enrollment snapshot
     * @param courses belonging to the snapshot
     * @return the constructed SemesterSnapshot object
     */
    public static SemesterSnapshot createSemesterSnapshot(String date, List<Course> courses) {

        return new SemesterSnapshot(date, courses);
    }

    /**
     * Creates a SemesterSnapshot from CSV records.
     *
     * @param date the date of the enrollment snapshot
     * @param rows the CSV records containing course information
     * @return the constructed SemesterSnapshot object
     */
    public static SemesterSnapshot createSemesterSnapshotFromRows(String date, List<String[]> rows) {
        List<Course> courses = createCoursesFromRows(rows);

        return createSemesterSnapshot(date, courses);
    }

    /**
     * Reads a CSV snapshot file and constructs a SemesterSnapshot.
     *
     * @param file the CSV snapshot file
     * @param date the date of the snapshot
     * @return the constructed SemesterSnapshot object
     * @throws IOException if the file cannot be read
     */
    public static SemesterSnapshot readSemesterSnapshot(File file, String date) throws IOException {
        List<String[]> rows = readSnapshotRows(file);

        return createSemesterSnapshotFromRows(date, rows);
    }

}
